# TeamPixPython – Python dependencies
#
# Install order matters: install PyTorch first, from the download.pytorch.org
# index that matches the GPU (InitPhotoProcessing.bat picks it automatically),
# then this file, then facenet-pytorch without its dependencies:
#
#   NVIDIA Turing or newer (incl. RTX 50xx), driver >= 580:
#     pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cu130
#   Other NVIDIA GPUs up to Hopper (e.g. Volta/TITAN V, or any pre-Blackwell
#   card on a driver < 580), driver >= 528:
#     pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cu126
#   No NVIDIA GPU:
#     pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements.txt
#   pip install --no-deps facenet-pytorch==2.6.0
#
# facenet-pytorch (the FaceNet face-recognition provider) is NOT listed here:
# it's unmaintained and declares torch<2.3 / torchvision<0.18 / Pillow<10.3,
# so installing it normally downgrades PyTorch to a CPU-only 2.2.x, which
# also has no GPU code for RTX 50xx cards. Its model code runs fine on
# current PyTorch, hence --no-deps. Its own runtime needs (requests, tqdm)
# are listed below.

rawpy>=0.18.0
opencv-python>=4.10.0
# EXIF capture-time reading for the Review page's burst grouping (culling_app.py).
# Already an indirect dependency of torchvision/facenet-pytorch, listed here
# explicitly since we now import it directly.
Pillow>=10.0.0
# Kept on NumPy 1.x: the tested combination. torch >= 2.3 also supports
# NumPy 2, so this can be lifted once mediapipe/basicsr/dlib are verified on
# it. mediapipe's declared opencv-contrib dependency wants numpy>=2 but works
# fine at runtime with 1.26.x.
numpy>=1.26.0,<2
scikit-learn>=1.3.0

# Default detection/pose/face-landmark engine (--engine mediapipe, the default).
# Permissively licensed (Apache-2.0), unlike the optional YOLO engine below.
# torchvision (installed alongside torch — see quick-start above) supplies the
# person-detector half of the hybrid body-detection pipeline (BSD-3-Clause).
mediapipe>=0.10.14

# Optional legacy engine (--engine yolo). Ultralytics YOLO models are
# AGPL-3.0 / GPL-3.0 licensed — see README licensing notes before enabling.
ultralytics>=8.3.0

# Optional but required when using the dlib face-recognition provider
face-recognition>=1.3.0
dlib>=19.24.0

# facenet-pytorch is installed separately with --no-deps (see the top of this
# file); these are its runtime dependencies (weight download + progress bar).
requests>=2.0.0
tqdm>=4.0.0

# Web UI for the end-to-end photo culling UI (culling_app.py), including its
# face-reco cluster tagging page. Permissive licenses (FastAPI MIT,
# uvicorn/starlette BSD-3-Clause, Jinja2 BSD-3-Clause).
fastapi>=0.110.0
uvicorn[standard]>=0.29.0
python-multipart>=0.0.9
jinja2>=3.1.0

# LLM-assisted burst culling (algo/llm/culling_provider.py OpenAIProvider,
# algo/stages/llm_culling.py). MIT licensed. Optional at runtime: the stage
# is skipped when no OpenAI API key is configured.
openai>=1.30.0

# Embeds each exported photo's culling "stars" rating into its EXIF
